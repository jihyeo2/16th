{{- $spotify := .Get "spotify" -}} {{- $youtube := .Get "youtube" -}} {{- $label := .Get "label" | default "Now Playing" -}} {{- if $spotify -}} {{- $kind := "track" -}} {{- $id := $spotify -}} {{- if in $spotify "spotify.com" -}} {{- $kind = $spotify | replaceRE `^.*open\.spotify\.com/(?:[a-z-]+/)?(track|album|playlist)/.*$` `$1` -}} {{- $id = $spotify | replaceRE `^.*open\.spotify\.com/(?:[a-z-]+/)?(?:track|album|playlist)/([A-Za-z0-9]+).*$` `$1` -}} {{- end -}}
{{ $label }} Spotify
{{- else if $youtube -}} {{- $id := $youtube -}} {{- if in $youtube "youtu" -}} {{- if in $youtube "watch?v=" -}} {{- $id = $youtube | replaceRE `^.*[?&]v=([A-Za-z0-9_-]+).*$` `$1` -}} {{- else -}} {{- $id = $youtube | replaceRE `^.*/([A-Za-z0-9_-]{6,})/?(?:\?.*)?$` `$1` -}} {{- end -}} {{- end -}}
{{ $label }} YouTube
{{- end -}}
